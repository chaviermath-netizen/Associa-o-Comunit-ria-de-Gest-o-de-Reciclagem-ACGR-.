from flask import Flask, render_template, redirect, url_for, request, flash
from flask_sqlalchemy import SQLAlchemy
from flask_login import LoginManager, UserMixin, login_user, login_required, logout_user, current_user
from werkzeug.security import generate_password_hash, check_password_hash

app = Flask(__name__)
app.config['SECRET_KEY'] = 'acgr-seguranca-mvp-2026-key'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///acgr.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)
login_manager = LoginManager()
login_manager.init_app(app)
login_manager.login_view = 'login'

# Modelo de Utilizador (Segurança e Autenticação)
class User(UserMixin, db.Model):
    id = db.Column(db.Integer, primary_key=True)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(200), nullable=False)

# Modelo de Chamado de Recolhimento
class Chamado(db.Model):
    id = db.Column(db.Integer, primary_key=True)
    solicitante = db.Column(db.String(120), nullable=False)
    material = db.Column(db.String(80), nullable=False)
    endereco = db.Column(db.String(250), nullable=False)
    status = db.Column(db.String(50), default='Pendente') # Pendente, Agendado, Concluído
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)

@login_manager.user_loader
def load_user(user_id):
    return User.query.get(int(user_id))

# Rota Inicial / Login (Tela 1)
@app.route('/', methods=['GET', 'POST'])
@app.route('/login', methods=['GET', 'POST'])
def login():
    if request.method == 'POST':
        email = request.form.get('email')
        password = request.form.get('password')
        user = User.query.filter_by(email=email).first()
        
        if user and check_password_hash(user.password_hash, password):
            login_user(user)
            return redirect(url_for('novo_chamado'))
        flash('Credenciais inválidas. Verifique seu e-mail e senha.', 'danger')
    return render_template('login.html')

# Rota de Cadastro de Chamado (Tela 2)
@app.route('/novo-chamado', methods=['GET', 'POST'])
@login_required
def novo_chamado():
    if request.method == 'POST':
        endereco = request.form.get('endereco')
        material = request.form.get('material')
        
        novo = Chamado(
            solicitante=current_user.email,
            material=material,
            endereco=endereco,
            status='Agendado',
            user_id=current_user.id
        )
        db.session.add(novo)
        db.session.commit()
        flash(f'Success! Collection request for ACGR-{novo.id+9000} has been scheduled.', 'success')
        return redirect(url_for('dashboard'))
    return render_template('novo_chamado.html')

# Rota do Painel e Resultados (Tela 3)
@app.route('/dashboard')
@login_required
def dashboard():
    chamados = Chamado.query.filter_by(user_id=current_user.id).all()
    total = len(chamados)
    pendentes = sum(1 for c in chamados if c.status == 'Pendente')
    agendados = sum(1 for c in chamados if c.status == 'Agendado')
    concluidos = sum(1 for c in chamados if c.status == 'Concluído')
    return render_template('dashboard.html', chamados=chamados, total=total, pendentes=pendentes, agendados=agendados, concluidos=concluidos)

@app.route('/logout')
@login_required
def logout():
    logout_user()
    return redirect(url_for('login'))

if __name__ == '__main__':
    with app.app_context():
        db.create_all()
        if not User.query.filter_by(email='morador.teste@acgr.local').first():
            hashed_pw = generate_password_hash('senha123', method='pbkdf2:sha256')
            user_teste = User(email='morador.teste@acgr.local', password_hash=hashed_pw)
            db.session.add(user_teste)
            db.session.commit()
    app.run(debug=True)
